# create file format
CREATE OR REPLACE FILE FORMAT healthcare_db.ingest.fileformat 
 TYPE = CSV 
 FIELD_DELIMITER = ',' 
 SKIP_HEADER = 1;


//-- ################################################################# --//
      2. Blob Service URL            
//-- ################################################################# --//

Storage Account -> settings -> Endpoints -> Blob Services:

healthcarestoragedata:
  -> settings 
  -> Endpoints 
  -> Blob Services:
  -> Blob Service Endpoint: https://healthcarestoragedata.blob.core.windows.net/


//-- ################################################################# --//
      4. Create stage in Snowflake
//-- ################################################################# --//

USE SCHEMA HEALTHCARE_DB.ingest;

# 01. based on storage integration
CREATE OR REPLACE STAGE healthcare_db.ingest.healthcare_stage_json
 STORAGE_INTEGRATION = HEALTHCARE_STORAGE_INTEGRATION 
 URL = 'azure://healthcarestoragedata.blob.core.windows.net/healthcare-container' 
 FILE_FORMAT = fileformat;

# 02. check stages
SHOW STAGES IN SCHEMA healthcare_db.ingest;


//-- ################################################################# --//
      5. Test stage in Snowflake
//-- ################################################################# --//

# 01. Upload file into storage container

# 02. Test data loading from stage into table

# make sure you can see the file
LIST @healthcare_db.ingest.healthcare_stage_json;

# inspect the staged data before loading it into a table
SELECT $1, $2, $3, $4, $5 FROM @healthcare_db.ingest.healthcare_stage_json;

# load file into landing table
COPY INTO HEALTHCARE_DB.INGEST.hospitals
  FROM(
      SELECT $1, $2, $3, $4, $5
      FROM @HEALTHCARE_DB.INGEST.healthcarestage_json 
)
FILE_FORMAT = (FORMAT_NAME = 'healthcare_db.ingest.fileformat');
