        last_updated_date as last_updated_date

//-- ################################################################# --//
     5. understand how incremental model works
         -- select updated data by a temporary view
         -- merge updated data into existing table
//-- ################################################################# --//

# 00. check dbt detailed log
START sql incremental model dbt_funnygorilla.stg_tpch_nations ........... [RUN]

# 01. create a temporary view 
create or replace temporary view PC_DBT_DB.dbt_funnygorilla.stg_tpch_nations__dbt_tmp
  as (
    with source as (
      select * from PC_DBT_DB.dbt_funnygorilla.nations
    ),
   renamed as (
      select
          n_nationkey       as nation_key,
          n_name            as name,
          n_regionkey       as region_key,
          last_updated_date as last_updated_date
      from source
    )
   select * from renamed
      -- this filter will only be applied on an incremental run
      where last_updated_date > (select max(last_updated_date) 
      from PC_DBT_DB.dbt_funnygorilla.stg_tpch_nations)
  )

# 02. merge clause
merge into PC_DBT_DB.dbt_funnygorilla.stg_tpch_nations as DBT_INTERNAL_DEST
        using PC_DBT_DB.dbt_funnygorilla.stg_tpch_nations__dbt_tmp as DBT_INTERNAL_SOURCE
        on ((DBT_INTERNAL_SOURCE.nation_key = DBT_INTERNAL_DEST.nation_key))
    
    when matched then update set
        "NATION_KEY"        = DBT_INTERNAL_SOURCE."NATION_KEY",
        "NAME"              = DBT_INTERNAL_SOURCE."NAME",
        "REGION_KEY"        = DBT_INTERNAL_SOURCE."REGION_KEY",
        "LAST_UPDATED_DATE" = DBT_INTERNAL_SOURCE."LAST_UPDATED_DATE"
    
    when not matched then insert
        ("NATION_KEY", "NAME", "REGION_KEY", "LAST_UPDATED_DATE")
        values
        ("NATION_KEY", "NAME", "REGION_KEY", "LAST_UPDATED_DATE")
