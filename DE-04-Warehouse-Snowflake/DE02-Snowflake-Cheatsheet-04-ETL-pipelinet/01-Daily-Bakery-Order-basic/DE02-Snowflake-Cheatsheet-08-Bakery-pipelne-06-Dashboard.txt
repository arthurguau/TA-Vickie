# Import python packages
import streamlit as st
import os
from snowflake.snowpark.context import get_active_session

# Write directly to the app
st.title(f"FGL - Bakery Order Summary")
st.write(
  """Bakery order summary is used by FGL training camp to develop data engineering techniques."""
)

# Create a database connection to Snowflake
# conn = st.connection("snowflake", type="snowflake")
# session = conn.session()

# Get the current credentials
session = get_active_session()

@st.cache_data
def load_table():
    #session = conn.session()
    return session.table("BAKERY_DB.ORDERS.SUMMARY_ORDERS").to_pandas() # to_pandas_batches

df = load_table()
row_count = df.shape[0]

#  Create an example dataframe
query = "SELECT BAKED_GOOD_TYPE, sum(TOTAL_QUANTITY) TOTAL\
         FROM BAKERY_DB.ORDERS.SUMMARY_ORDERS \
         group by BAKED_GOOD_TYPE \
         order by TOTAL"
order_df = session.sql(query).to_pandas()

# Create a simple bar chart
st.subheader("Bar: Total Quantity Per Bakery Type")
st.bar_chart(data=order_df, x="BAKED_GOOD_TYPE", y="TOTAL")

# Finally, display the contents of the table.
st.subheader(f"Table: Underlying Bakery Order Data  -  Total: {row_count}")
st.dataframe(df, use_container_width=True)
