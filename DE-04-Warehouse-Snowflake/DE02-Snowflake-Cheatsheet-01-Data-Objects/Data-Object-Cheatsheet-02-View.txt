Snowflake account created


//-- ################################################################# --//
     0. Set up database                        
//-- ################################################################# --//

-- 00. login to Snowflake and open a worksheet
SHOW databases;
SHOW warehouses;
SHOW users;
SHOW roles;

-- 01. Setup a transient DB (using a data warehouse?) 
USE ROLE SYSADMIN;
USE WAREHOUSE COMPUTE_WH; 

CREATE OR REPLACE DATABASE FUNNY_DB 
Comment = 'Database for Funnygorilla';

-- 02. Switch to ACCOUNTADMIN role to view databases  
USE ROLE ACCOUNTADMIN;
SHOW DATABASES;

-- 03. Alter database to have 10 day data retention time
USE ROLE SYSADMIN;
ALTER DATABASE FUNNY_TRANSIENT_DB  
SET DATA_RETENTION_TIME_IN_DAYS=10;

CREATE OR REPLACE DATABASE FUNNY_DB; 
ALTER DATABASE FUNNY_DB  
SET DATA_RETENTION_TIME_IN_DAYS=10;

-- 04. create table
USE ROLE SYSADMIN;
CREATE OR REPLACE TABLE FUNNY_TRANSIENT_DB.PUBLIC.SUMMARY 
   (CASH_AMT number,
    RECEIVABLES_AMT number,
    CUSTOMER_AMT number);


//-- ################################################################# --//
     2. Set up schema & table                         
//-- ################################################################# --//

USE ROLE SYSADMIN; 
USE DATABASE FUNNY_DB;

-- 01. Create schema
CREATE OR REPLACE SCHEMA BANKING;

SHOW SCHEMAS; 

-- 02. Create tables
CREATE OR REPLACE TABLE CUSTOMER_ACCT (Customer_Account int, Amount int, transaction_ts timestamp);
CREATE OR REPLACE TABLE CASH (Customer_Account int, Amount int, transaction_ts timestamp);
CREATE OR REPLACE TABLE RECEIVABLES (Customer_Account int, Amount int, transaction_ts timestamp);

SHOW TABLES; 


//-- ################################################################# --//
     2. non-materialized view           
//-- ################################################################# --//

-- NONMATERIALIZED VIEW
USE ROLE SYSADMIN;
CREATE OR REPLACE VIEW FUNNY_DB.PUBLIC.NEWVIEW AS
  SELECT CC_NAME 
  FROM (SELECT * FROM SNOWFLAKE_SAMPLE_DATA.TPCDS_SF100TCL_OLD.CALL_CENTER);



//-- ################################################################# --//
     3. materialized  view               
//-- ################################################################# --//

-- MATERIALIZED VIEW (requires enterprise edtion)
USE ROLE SYSADMIN;

CREATE OR REPLACE MATERIALIZED VIEW FUNNY_DB.BANKING.SUMMARY_MVW AS
   SELECT * FROM (SELECT * FROM FUNNY_DB.BANKING.SUMMARY);

USE SCHEMA FUNNY_DB.PUBLIC;
SHOW VIEWS;


//-- ################################################################# --//
     Clean up  
      - database

-- Cleanup the sample db, user and roles

drop database FUNNY_TRANSIENT_DB;
drop database FUNNY_DB;
