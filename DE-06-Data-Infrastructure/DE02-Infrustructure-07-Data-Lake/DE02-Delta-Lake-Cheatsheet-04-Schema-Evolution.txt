//-- ################################################################# --//
     3. Schema Evolution
//-- ################################################################# --//

# 01. create a Delta table with a simple schema
df = spark.createDataFrame([("bob", 47), ("mike", 23), ("tom", 51)]).toDF("first_name", "age")
df.write.format("delta").save("test01/toy_data")

df = spark.createDataFrame([("Peter", 47), ("li", 23), ("leonard", 51)]).toDF("first_name", "age")

# write mode: append, overwrite
df.write.format("delta").mode("append").save("test01/toy_data") 
  
spark.read.format("delta").load("test01/toy_data").show()

# 02. try to write data with a different schema to this same Delta table:
df_new = spark.createDataFrame([("frank", 68, "usa"), ("jordana", 26, "brasil")]).toDF("first_name", "age", "country")

df_new.write.format("delta").mode("append").save("test01/toy_data")


//-- ################################################################# --//
     Update Evolution
//-- ================================================================= --//

# update schema
df_new.write.option("mergeSchema", "true").mode("append").format("delta").save("test01/toy_data")

# show new changed table
spark.read.format("delta").load("test01/toy_data").show()
