# 01. create storage integration
  -- storage url: ('azure://<account>.blob.core.windows.net/<container>/<path>/');
CREATE STORAGE INTEGRATION <integration_name>
  TYPE = EXTERNAL_STAGE
  STORAGE_PROVIDER = <cloud_provider> -- AWS, AZURE, or GCS
  ENABLED = TRUE
  AZURE_TENANT_ID = '<tenant-id>'
  STORAGE_ALLOWED_LOCATIONS = ('<cloud_storage_url>', ...);

CREATE STORAGE INTEGRATION HEALTHCARE_STORAGE_INTEGRATION
  TYPE = EXTERNAL_STAGE
  STORAGE_PROVIDER = AZURE
  ENABLED = TRUE
  AZURE_TENANT_ID = 'b97c54b4-4655-48e1-bf4e-c5570779fb48'
  STORAGE_ALLOWED_LOCATIONS = ('azure://healthcarestoragedata.blob.core.windows.net/healthcare-container');

# 02. check storage integration created
SHOW INTEGRATIONS 

# drop storage-integration - if you need to drop integrations
DROP INTEGRATION [IF EXISTS] <integration_name>;
