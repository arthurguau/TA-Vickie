-- create sales transaction table which combines the SALES_STREAM with the product table 
   adds a timestamp so that we can know when the sales data was recorded:

CREATE OR REPLACE TABLE FUNNY_TASK_DB.TASKSDEMO.SALES_TRANSACT
   (Prod_ID int,
    Prod_Desc varchar(),
    Category varchar(30),
    Segment varchar(20),
    Mfg_ID int,
    Mfg_Name varchar(50),
    Customer varchar(),
    Zip varchar(),
    Qty int,
    Revenue decimal (10, 2),
    TS timestamp);

-- 02. mannual add some raws into transact talbe
INSERT INTO FUNNY_TASK_DB.TASKSDEMO.SALES_TRANSACT
        (Prod_ID,Prod_Desc,Category,Segment,Mfg_Id,Mfg_Name,Customer,Zip,Qty,Revenue,TS)
  SELECT
    s.Prod_ID,
    p.Prod_Desc,
    p.Category,
    p.Segment,
    p.Mfg_ID,
    p.Mfg_Name,
    s.Customer,
    s.Zip,
    s.Qty,
    s.Revenue,
    current_timestamp
  FROM
    FUNNY_TASK_DB.TASKSDEMO.SALES_STREAM s JOIN FUNNY_TASK_DB.TASKSDEMO.PRODUCT p ON s.Prod_ID = p.Prod_ID;


SELECT * FROM FUNNY_TASK_DB.TASKSDEMO.SALES_TRANSACT;


//-- ################################################################# --//
      4. Create task                      
//-- ################################################################# --//

-- 01. create a task

CREATE OR REPLACE TASK FUNNY_TASK_DB.TASKSDEMO.SALES_TASK
     WAREHOUSE = compute_wh 
     SCHEDULE  = '1 minute'
WHEN system$stream_has_data('FUNNY_TASK_DB.TASKSDEMO.SALES_STREAM')
AS
INSERT INTO
    FUNNY_TASK_DB.TASKSDEMO.SALES_TRANSACT
        (Prod_ID,Prod_Desc,Category,Segment,Mfg_Id,Mfg_Name,Customer,Zip,Qty,Revenue,TS)
SELECT
    s.Prod_ID,
    p.Prod_Desc,
    p.Category,
    p.Segment,
    p.Mfg_ID,
    p.Mfg_Name,
    s.Customer,
    s.Zip,
    s.Qty,
    s.Revenue,
    current_timestamp
FROM
    FUNNY_TASK_DB.TASKSDEMO.SALES_STREAM s JOIN FUNNY_TASK_DB.TASKSDEMO.PRODUCT p ON s.Prod_ID = p.Prod_ID;

-- 02. check create task

-- 03. test and execute task a single time
use role sysadmin;
execute task SALES_TASK;

-- 04. view previous and scheduled task executions
select *
  from table(information_schema.task_history())
  order by scheduled_time desc;

-- 05. put task into action
alter task SALES_TASK resume;

-- 06. resume task (as inactive by default)
ALTER TASK FUNNY_TASK_DB.TASKSDEMO.SALES_TASK RESUME;
