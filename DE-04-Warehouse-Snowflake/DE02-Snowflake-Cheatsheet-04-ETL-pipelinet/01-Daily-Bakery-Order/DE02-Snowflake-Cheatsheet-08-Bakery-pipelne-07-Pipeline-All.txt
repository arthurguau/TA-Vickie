      3. Create infrastructure tables for 3 logical tiers:
          - staging table
          - warehousing tables
          - delivery tables                 
//-- ################################################################# --//

use database BAKERY_DB;
use schema ORDERS;

-- 01. Create stage table
create table ORDERS_STG (
  customer         varchar,
  order_date       date,
  delivery_date    date,
  baked_good_type  varchar,
  quantity         number,
  source_file_name varchar,
  load_ts          timestamp
);

-- 02. Create warehousing table
create table CUSTOMER_ORDERS (
  customer varchar,
  order_date date,
  delivery_date date,
  baked_good_type varchar,
  quantity number,
  source_file_name varchar,
  load_ts timestamp
);

-- 03. Create delivery table [summary table]
create table SUMMARY_ORDERS(
  delivery_date date,
  baked_good_type varchar,
  total_quantity number
);

//-- ################################################################# --//
      5. Load file into staging table  
          - copy command                           
//-- ################################################################# --//


use database BAKERY_DB;
use schema ORDERS;

-- 01. Use copy command to load into table
copy into ORDERS_STG
from (
  select $1, $2, $3, $4, $5, metadata$filename, current_timestamp() 
  from @ORDERS_STAGE
)
file_format = (type = CSV, skip_header = 1)
on_error = abort_statement
purge = true;   -- purge data

-- 02. Check data in table
select * from ORDERS_STG;

-- 03. Check stage
list @ORDERS_STAGE;


//-- ################################################################# --//
      4. Merge data into warehousing table                             
//-- ################################################################# --//

use database BAKERY_DB;
use schema ORDERS;

-- merge into the target table
merge into CUSTOMER_ORDERS tgt

-- the source table
using ORDERS_STG as src 

-- the columns that ensure uniqueness
on src.customer = tgt.customer 
  and src.delivery_date = tgt.delivery_date 
  and src.baked_good_type = tgt.baked_good_type

-- update the target table with the values from the source table
when matched then 
  update set tgt.quantity = src.quantity, 
             tgt.source_file_name = src.source_file_name, 
             tgt.load_ts = current_timestamp()

-- insert new values from the source table into the target table
when not matched then
  insert (customer, 
          order_date, 
          delivery_date, 
          baked_good_type, 
          quantity, 
          source_file_name, 
          load_ts)
  values(src.customer, 
         src.order_date, 
         src.delivery_date, 
         src.baked_good_type, 
         src.quantity, 
         src.source_file_name,
         current_timestamp());

-- check table
select * from CUSTOMER_ORDERS order by delivery_date desc;


//-- ################################################################# --//
      5. Transform data          
//-- ################################################################# --//

-- 01. Define summary logic
select delivery_date, baked_good_type, sum(quantity) as total_quantity
  from CUSTOMER_ORDERS
  group by all;

-- 02. remove all rows from table
truncate table SUMMARY_ORDERS;

-- 03. insert summary data into summary table
insert into SUMMARY_ORDERS(delivery_date, baked_good_type, total_quantity)
  select delivery_date, baked_good_type, sum(quantity) as total_quantity
  from CUSTOMER_ORDERS
  group by all;

-- create a task admin role
USE ROLE SECURITYADMIN;
CREATE ROLE TASKADMIN;

-- grant privilege
USE ROLE ACCOUNTADMIN;
GRANT EXECUTE TASK, EXECUTE MANAGED TASK ON ACCOUNT TO ROLE TASKADMIN;

-- grant role TASKADMIN TO ROLE SYSADMIN;
USE ROLE SECURITYADMIN;
GRANT ROLE TASKADMIN TO ROLE SYSADMIN;



use database BAKERY_DB;
use schema ORDERS; 

-- 01. create a task
create task PROCESS_ORDERS
  warehouse = BAKERY_WH
  schedule = '10 M'
as
begin
  -- staging table to warehousing table
  truncate table ORDERS_STG;
  copy into ORDERS_STG
     from (        
        select $1, $2, $3, $4, $5, metadata$filename, current_timestamp() 
        from @ORDERS_STAGE
     )
     file_format = (type = CSV, skip_header = 1)
     on_error = abort_statement
     purge = true;

  merge into CUSTOMER_ORDERS tgt
    -- the source table
    using ORDERS_STG as src 

    -- the columns that ensure uniqueness
    on src.customer = tgt.customer 
       and src.delivery_date = tgt.delivery_date 
       and src.baked_good_type = tgt.baked_good_type

     -- update the target table with the values from the source table
     when matched then 
     update set tgt.quantity = src.quantity, 
             tgt.source_file_name = src.source_file_name, 
             tgt.load_ts = current_timestamp()

     -- insert new values from the source table into the target table
     when not matched then
         insert (customer, 
            order_date, 
            delivery_date, 
            baked_good_type, 
            quantity, 
            source_file_name, 
            load_ts)
         values(src.customer, 
            src.order_date, 
            src.delivery_date, 
            src.baked_good_type, 
            src.quantity, 
            src.source_file_name,
            current_timestamp());

  truncate table SUMMARY_ORDERS;

  -- warehousing table to delivery table
  insert into SUMMARY_ORDERS(delivery_date, baked_good_type, total_quantity)
     select delivery_date, baked_good_type, sum(quantity) as total_quantity
     from CUSTOMER_ORDERS
     group by all;

end;

-- 02. check create task

-- 03. test and execute task a single time
use role sysadmin;
execute task PROCESS_ORDERS;

-- 04. view previous and scheduled task executions
select *
  from table(information_schema.task_history())
  order by scheduled_time desc;

-- 05. put task into action
alter task PROCESS_ORDERS resume;


//-- ################################################################# --//
      7. Clean up                
//-- ################################################################# --//

ALTER TASK BAKERY_DB.ORDERS.PROCESS_ORDERS SUSPEND;

DROP DATABASE BAKERY_DB;
DROP WAREHOUSE BAKERY_WH;
