# concept: With Statement in Python
simplifies resource management by automatically handling setup and cleanup, 
ensuring files or connections close safely even if errors occur.

Replaces long try-except–finally blocks with cleaner syntax.
Improves readability by reducing unnecessary boilerplate code.

# sample try-finally

try:
    file.write("Hello, Python!")

except ZeroDivisionError:
    # 2. Except: Runs ONLY if the specific error occurs
    print("Error: You cannot divide by zero!")

else:

finally:
    file.close()  # Ensure file is closed

import os
from dotenv import load_dotenv
import json
import pathlib
import airflow
import requests
import requests.exceptions as request_exceptions
from datetime import date
from datetime import datetime, timedelta

from airflow import DAG
from airflow.providers.standard.operators.bash import BashOperator
from airflow.providers.standard.operators.python import PythonOperator
from airflow.sdk import task

dag_owner = 'Kendrick'

load_dotenv()

# 03. get image

def _get_pictures():
    pathlib.Path("./tmp/images").mkdir(parents=True, exist_ok=True)
    api_key = os.getenv("NASA_KEY") 
    url = f'https://api.nasa.gov/planetary/apod?api_key={api_key}'
    response = requests.get(url).json()
    today_image = response['hdurl']
    with open(f'todays_image_{date.today()}.png', 'wb') as f:
        f.write(requests.get(today_image).content)

 python3 ./dags/MyfirstDAG.py

# 04. default values
default_args = {
        'owner': FGL_Vickie,
        'depends_on_past': False,
        'retries': 2,
        'retry_delay': timedelta(minutes=5)
        }


# 05. pipeline definition
with DAG(dag_id='download_ASOD_image',
        default_args=default_args,
        description='download and notify ',
        start_date=datetime(2026, 5, 23),
        schedule='@daily',
        catchup=False,
        tags=['None']
):

   get_pictures = PythonOperator(
      task_id="get_pictures",
      python_callable=_get_pictures,
   )

   notify = BashOperator(
       task_id="notify",
       bash_command='echo f"Image for today has been added!"',
   )

# 06. Define task dependency
   get_pictures >> notify
