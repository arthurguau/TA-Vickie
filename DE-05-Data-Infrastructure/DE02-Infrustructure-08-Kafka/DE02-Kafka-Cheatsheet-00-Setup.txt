//-- ################################################################# --//
     2. Create Topics
        -- from kafka webUI
        -- from Kafka container
//-- ################################################################# --//

# 00.access Kafka container
docker exec -it kafka1 bash
cd /opt/bitnami/kafka/bin
cd /usr/bin/

# 01. list topics
./kafka-topics.sh --bootstrap-server localhost:9092 --list

# 02. create topic 'funnygorilla-topic'
./kafka-topics.sh --bootstrap-server localhost:9092 --create --topic funnygorilla-topic --partitions 1 --replication-factor 1

# 03. list topics
./kafka-topics.sh --bootstrap-server localhost:9092 --list


//-- ################################################################# --//
