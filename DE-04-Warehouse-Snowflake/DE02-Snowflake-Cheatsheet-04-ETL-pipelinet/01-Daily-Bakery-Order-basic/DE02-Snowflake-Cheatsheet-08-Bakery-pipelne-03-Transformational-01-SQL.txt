//-- ################################################################# --//
      1. Source model to Staging model                           
//-- ################################################################# --//

user role accountadmin;
use warehouse BAKERY_WH;
use database BAKERY_DB;
use schema ORDERS;

-- 01. select all from file and add file name and loading timestamp
select $1, $2, $3, $4, $5, 
     metadata$filename, 
     current_timestamp() 
  from @ORDERS_STAGE

# 02. Add copy into command 
copy into ORDERS_STG
  from (        
        select $1, $2, $3, $4, $5, metadata$filename, current_timestamp() 
        from @ORDERS_STAGE
  )
  file_format = (type = CSV, skip_header = 1)
  on_error = abort_statement
  purge = true;

-- 03. Check data in table
select * from ORDERS_STG;

-- 04. Check stage
list @ORDERS_STAGE;


//-- ################################################################# --//
      2. Staging model to warehousing model
          - merge clause                             
//-- ################################################################# --//

-- merge into the target table
merge into CUSTOMER_ORDERS tgt

-- the source table
using ORDERS_STG as src 

-- the columns that ensure uniqueness
on src.customer = tgt.customer 
  and src.delivery_date = tgt.delivery_date 
  and src.baked_good_type = tgt.baked_good_type

-- update the warehousing table with the values from the source table
when matched then 
  update set tgt.quantity = src.quantity, 
             tgt.source_file_name = src.source_file_name, 
             tgt.load_ts = current_timestamp()

-- insert new values from the source table into the warehousing table
when not matched then
  insert (customer, 
          order_date, 
          delivery_date, 
          baked_good_type, 
          quantity, 
          source_file_name, 
          load_ts)
  values(src.customer, 
         src.order_date, 
         src.delivery_date, 
         src.baked_good_type, 
         src.quantity, 
         src.source_file_name,
         current_timestamp());


//-- ################################################################# --//
      3. Warehousing model to delivery model      
//-- ################################################################# --//

-- 01. Native sql 
INSERT INTO target_table (delivery_date, baked_good_type, total_quantity)
  select delivery_date, baked_good_type, sum(quantity) as total_quantity
  from CUSTOMER_ORDERS
  group by all;
