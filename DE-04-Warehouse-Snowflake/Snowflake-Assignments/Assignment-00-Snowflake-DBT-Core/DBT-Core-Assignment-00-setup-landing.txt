USE ROLE ACCOUNTADMIN;

-- 01. Create warehouse
CREATE OR REPLACE WAREHOUSE portfolio_dev_wh  WITH WAREHOUSE_SIZE = 'XSMALL' AUTO_SUSPEND = 60 AUTO_RESUME = TRUE MIN_CLUSTER_COUNT = 1 MAX_CLUSTER_COUNT = 1 INITIALLY_SUSPENDED = TRUE;


//-- ################################################################# --//
     3. Create landing zone: (Snowflake): up to ingestion process to feed data
          - database
          - schema
          - table                        
//-- ################################################################# --//

USE ROLE ACCOUNTADMIN;

-- 01. Create database
USE ROLE ACCOUNTADMIN;
CREATE DATABASE IF NOT EXISTS PORTFOLIO_TRACKING
    COMMENT = 'DB for the portfolio tracking project';



-- 02. Create schema
USE ROLE ACCOUNTADMIN;
CREATE SCHEMA PORTFOLIO_TRACKING.SOURCE_DATA;

-- 03. Create table
CREATE OR REPLACE TABLE
  PORTFOLIO_TRACKING.SOURCE_DATA.ABC_BANK_POSITION (
    accountID         TEXT not null,
    symbol            TEXT not null unique,
    description       TEXT not null,
    exchange          TEXT not null,
    report_date       DATE not null,
    quantity          NUMBER(38,0) not null,
    cost_base         NUMBER(38,5) not null,
    position_value    NUMBER(38,5) not null,
    currency          TEXT not null
); 


//-- ################################################################# --//
     4. Load file into Snowflake - emulate ingestion process to feed data  
         01. Define file format
         02. Load data into landing table                       
//-- ################################################################# --//

//-- ################################################################# --//
     01. Define file format (Snowflake)                            
//-- ================================================================= --//

# 01. define a file format for loading external data - csv file

USE ROLE ACCOUNTADMIN;

CREATE FILE FORMAT
  PORTFOLIO_TRACKING.SOURCE_DATA.ABC_BANK_CSV_FILE_FORMAT
    TYPE = 'CSV'
        COMPRESSION = 'AUTO'
        FIELD_DELIMITER = ','
        RECORD_DELIMITER = '\n'
        SKIP_HEADER = 1
        FIELD_OPTIONALLY_ENCLOSED_BY = '\042'
        TRIM_SPACE = FALSE
        ERROR_ON_COLUMN_COUNT_MISMATCH = TRUE
        ESCAPE = 'NONE'
        ESCAPE_UNENCLOSED_FIELD = '\134'
        DATE_FORMAT = 'AUTO'
        TIMESTAMP_FORMAT = 'AUTO'
        NULL_IF = ('\\N')
;

# 02. Check snowflake under PORTFOLIO_TRACKING.SOURCE_DATA
