//-- ################################################################# --//
     01. Post a Student                           
//-- ################################################################# --//

# 01. Setup request message and send
http method - Post
url - http://localhost:8087/studentmanagement/api/v1/students/
Request body - {"firstName":"Michael","lastName":"Gu","contact":"arthur@hotmail.com","course":"CS"}


# 02. check response message
Scripts -> post-response:

# status code - 201
pm.test("check response code",  function (){
    pm.response.to.have.status(201);
})

# content type - application/json
pm.test("check response code",  function (){
    pm.response.headers.get('content-type');
})

# response property

pm.test("check response code",  function (){
  var jsonData = pm.response.json();
  pm.expect(pm.response.text()).to.include("id");
})

//-- ################################################################# --//
     02. Read a Student                           
//-- ################################################################# --//

# 01. Setup request message and send
http method - Get
url - http://localhost:8087/studentmanagement/api/v1/students/1
Request body - empty


# 02. check response message

# status code - 200
pm.response.to.have.status(200);

# content type - application/json
pm.response.to.have.header('Content-Type', 'application/json');

# response property
var jsonData = pm.response.json();
pm.expect(jsonData.value).to.eql(100);

pm.expect(pm.response.text()).to.include("id");
