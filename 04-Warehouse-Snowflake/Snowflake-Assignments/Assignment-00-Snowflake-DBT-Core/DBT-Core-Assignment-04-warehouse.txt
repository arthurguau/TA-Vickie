# 01. create file REF_POSITION_ABC_BANK.sql into ./models/warehouse/
# 02. setup connection to staging model by 'ref'

SELECT
    *
    , POSITION_VALUE - COST_BASE as UNREALIZED_PROFIT
    , ROUND(UNREALIZED_PROFIT / COST_BASE, 5)
        as UNREALIZED_PROFIT_PCT
FROM {{ ref('STG_ABC_BANK_POSITION') }}


//-- ################################################################# --//
     2. Deploy model                        
//-- ################################################################# --//

# 01. Deploy model
dbt clean &&  dbt build

# 02. Check target folder
tree ./target

# 03. Deploy model
dbt run -s warehouse
